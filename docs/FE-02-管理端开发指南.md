        header: "操作",

            h(Button, { variant: "ghost", size: "sm", onClick: () => viewDetail(row.original.id) }, () => "查看"),
            h(Button, { variant: "ghost", size: "sm", onClick: () => editMember(row.original.id) }, () => "编辑"),

    get data() { return data.value?.list ?? [] },

## 5. 权限控制

### 5.1 Composable

```typescript
// src/composables/usePermission.ts
import { computed } from "vue";
import { useAuthStore } from "@/stores/auth";

export function usePermission() {
    const authStore = useAuthStore();

    const hasPermission = (code: string): boolean => {
        return authStore.permissions.includes(code);
    };

    const hasAnyPermission = (codes: string[]): boolean => {
        return codes.some(code => hasPermission(code));
    };

    return { hasPermission, hasAnyPermission };
}
```

### 5.2 组件级权限指令

```typescript
// src/directives/permission.ts
import { Directive } from "vue";
import { useAuthStore } from "@/stores/auth";

export const vPermission: Directive<HTMLElement, string | string[]> = {
    mounted(el, binding) {
        const store = useAuthStore();
        const required = Array.isArray(binding.value) ? binding.value : [binding.value];
        const allowed = required.some(code => store.permissions.includes(code));
        if (!allowed) el.parentNode?.removeChild(el);
    },
};
```

使用：

```vue
<Button v-permission="'party:member:create'">新增党员</Button>
<Button v-permission="['party:member:delete', 'party:admin']">删除</Button>
```

### 5.3 路由守卫

```typescript
// src/router.ts
router.beforeEach(async (to) => {
    const authStore = useAuthStore();

    if (!authStore.isAuthenticated && to.name !== "login") {
        return { name: "login" };
    }

    // 检查路由元信息中的权限要求
    if (to.meta.requiredPermission) {
        if (!authStore.hasPermission(to.meta.requiredPermission)) {
            return { name: "403" };
        }
    }
});
```

## 6. API 类型安全

### 6.1 OpenAPI 自动生成

后端使用 `utoipa` 自动生成 OpenAPI 3.0 规范文件。前端通过 `openapi-typescript` 生成类型：

```bash
npx openapi-typescript http://localhost:8080/api/docs/openapi.json \
  --output src/api/generated/schema.ts
```

### 6.2 类型化请求封装

```typescript
// src/api/index.ts
import axios from "axios";
import type { paths } from "./generated/schema";
import createClient from "openapi-fetch";

const { GET, POST } = createClient<paths>({ baseUrl: "/api/v1" });

export const api = {
    party: {
        getMembers(params: PageReq) {
            return GET("/party/members", { params }).then(res => res.data);
        },
        getMember(id: string) {
            return GET("/party/members/{id}", { params: { path: { id } } }).then(res => res.data);
        },
    },
};
```

## 7. 多租户上下文

```typescript
// src/composables/useTenantContext.ts
export function useTenantContext() {
    const tenantStore = useTenantStore();

    // 当前租户信息
    const tenantName = computed(() => tenantStore.currentTenant?.name);
    const tenantLogo = computed(() => tenantStore.currentTenant?.logo_url);

    // 功能开关检查
    const isFeatureEnabled = (feature: string): boolean => {
        return tenantStore.featureFlags.includes(feature);
    };

    return { tenantName, tenantLogo, isFeatureEnabled };
}
```

## 8. SSE 实时推送集成

```typescript
// src/composables/useSSE.ts
export function useSSE(onEvent: (type: string, data: any) => void) {
    let source: EventSource | null = null;
    let reconnectTimer: number;

    const connect = () => {
        const token = useAuthStore().token;
        source = new EventSource(`/api/v1/sse/subscribe?token=${token}`);

        source.addEventListener("notification", (e) => onEvent("notification", JSON.parse(e.data)));
        source.addEventListener("approval_status", (e) => onEvent("approval_status", JSON.parse(e.data)));

        source.onerror = () => {
            source?.close();
            reconnectTimer = window.setTimeout(connect, 5000); // 5s 后重连
        };
    };

    onMounted(connect);
    onUnmounted(() => {
        source?.close();
        clearTimeout(reconnectTimer);
    });
}
```
