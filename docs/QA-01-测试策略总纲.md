pub async fn setup_test_env() -> TestContext {

### 3.2 API 集成测试示例

```rust
// server/tests/api/party_member_test.rs
#[tokio::test]
async fn test_create_party_member_success() {
    let ctx = setup_test_env().await;
    let token = create_test_token(&ctx, tenant_a_id, user_admin).await;

    let response = ctx.app
        .clone()
        .oneshot(
            Request::builder()
                .method("POST")
                .uri("/api/v1/party/members")
                .header("Authorization", format!("Bearer {token}"))
                .header("Content-Type", "application/json")
                .body(Body::from(json!({
                    "real_name": "张三",
                    "org_id": org_id,
                    "member_type": "formal"
                }).to_string()))
                .unwrap(),
        )
        .await
        .unwrap();

    assert_eq!(response.status(), StatusCode::CREATED);
}

#[tokio::test]
async fn test_cross_tenant_access_denied() {
    let ctx = setup_test_env().await;
    // 创建租户 A 的党员
    let member_a = create_member_in_tenant(&ctx, tenant_a_id).await;
    // 用租户 B 的 Token 访问租户 A 的数据
    let token_b = create_test_token(&ctx, tenant_b_id, user_b).await;

    let response = ctx.app
        .clone()
        .oneshot(
            Request::builder()
                .method("GET")
                .uri(format!("/api/v1/party/members/{}", member_a.id))
                .header("Authorization", format!("Bearer {token_b}"))
                .body(Body::empty())
                .unwrap(),
        )
        .await
        .unwrap();

    assert_eq!(response.status(), StatusCode::FORBIDDEN);
    assert_eq!(get_error_code(response).await, "40301");
}
```

// e2e/admin/login.spec.ts
import { test, expect } from '@playwright/test';

test('P0: admin login and view dashboard', async ({ page }) => {
  await page.goto('https://admin.test.example.com');
  await page.fill('[data-testid=username]', 'admin');
  await page.fill('[data-testid=password]', 'Test@123456');
  await page.click('[data-testid=login-btn]');
  await expect(page.locator('[data-testid=dashboard]')).toBeVisible();

// e2e/admin/cross-tenant.spec.ts
test('Security: tenant A cannot see tenant B data', async ({ request }) => {
  const tokenA = await getTenantAToken(request);
  const response = await request.get('/api/v1/orgs/tree', {
    headers: { Authorization: `Bearer ${tokenA}` }
  });
  const data = await response.json();
  // 确保返回的组织树只包含租户 A 的数据
  for (const org of data.data.list) {
    expect(org.tenant_id).toBe(TENANT_A_ID);
  }
});

## 5. CI/CD 门禁

### 5.1 Pipeline 阶段

```yaml
# .github/workflows/ci.yml
stages:
  - lint:
      - cargo clippy -- -D warnings
      - cargo fmt --check
  - unit-test:
      - cargo test --workspace --lib
      - coverage check: >= threshold per crate
  - build:
      - cargo build --release
      - docker build
  - integration-test:
      - cargo test --workspace --test '*'
      - requires: PostgreSQL + Redis containers
  - security-scan:
      - cargo audit
      - trivy image scan
  - deploy-staging:
      - helm upgrade --install
  - e2e-test:
      - npx playwright test
      - miniprogram automator tests
  - deploy-production:
      - manual approval required
```

### 5.2 发布门禁规则

| 检查项 | 条件 | 不通过时 |
|--------|------|----------|
| Lint | clippy 无 warning | 阻止合并 |
| 单元测试 | 全部通过 + 覆盖率达标 | 阻止合并 |
| 集成测试 | 全部通过（含跨租户越权） | 阻止合并 |
| 安全扫描 | 无高危漏洞 | 阻止部署 |
| E2E | P0 用例全部通过 | 阻止发布 |
| Code Review | 至少 1 人批准 | 阻止合并 |
