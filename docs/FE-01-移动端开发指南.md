## 2. Cool Unix 核心概念

### 2.1 Service 层（统一 API 请求）

```typescript
// src/service/party.ts
import { useService } from '@/cool';

export default () => {
    const { request } = useService();

    return {
        // 获取党员列表
        getMembers(params: PageReq & { org_id?: string }) {
            return request<{ list: PartyMember[]; total: number }>({
                url: '/api/v1/party/members',
                method: 'GET',
                params,
            });
        },

        // 创建党员档案
        createMember(data: CreatePartyMemberDto) {
            return request<PartyMember>({
                url: '/api/v1/party/members',
                method: 'POST',
                data,
            });
        },
    };
};
```

### 2.2 Store（Pinia 状态管理）

```typescript
// src/store/modules/user.ts
import { defineStore } from 'pinia';
import partyService from '@/service/party';

export const useUserStore = defineStore('user', {
    state: () => ({
        userInfo: null as UserInfo | null,
        token: '',
        tenantId: '',
    }),

    actions: {
        async loginByWechat(code: string) {
            const res = await partyService.wechatLogin({ code });
            this.token = res.token;
            this.userInfo = res.user;
            this.tenantId = res.tenant_id;
        },

        logout() {
            this.token = '';
            this.userInfo = null;
            uni.reLaunch({ url: '/pages/index/index' });
        },
    },
});
```

## 3. 组件使用规范

### 3.1 Cool Unix 内置组件优先

优先使用 `cool-ui` 内置组件，不足时再自定义：

| 场景 | 推荐组件 |
|------|----------|
| 列表页 | ListView / ListViewRefresh |
| 表单 | Form + Input / Select / Cascader / Upload |
| 详情展示 | Text / Tag / Avatar / ReadMore |
| 操作反馈 | Toast / Confirm / ActionSheet / Popup |
| 数据展示 | Banner / Timeline / Progress / Skeleton |
| 特色功能 | Qrcode(扫码签到) / Sign(手写签名) / SlideVerify |

### 3.2 页面模板示例：党员详情

```vue
<template>
    <cl-page>
        <cl-topbar :title="member.real_name" back />

        <view class="p-4 space-y-4">
            <!-- 基本信息 -->
            <cl-card title="基本信息">
                <template #default>
                    <view class="flex items-center gap-3">
                        <cl-avatar :src="member.avatar_url" size="lg" />
                        <view class="flex-1">
                            <text class="text-lg font-bold">{{ member.real_name }}</text>
                            <view class="flex gap-2 mt-1">
                                <cl-tag>{{ memberTypeLabel }}</cl-tag>
                                <cl-tag type="info">{{ member.position || '普通党员' }}</cl-tag>
                            </view>
                        </view>
                    </view>
                    <view class="mt-4 grid grid-cols-2 gap-y-2 text-sm">
                        <text class="text-gray-500">入党时间</text>
                        <text>{{ member.join_date }}</text>
                        <text class="text-gray-500">所属组织</text>
                        <text>{{ member.org_name }}</text>
                    </view>
                </template>
            </cl-card>

            <!-- 党费记录 -->
            <cl-card title="党费记录">
                <template #default>
                    <cl-timeline :data="duesRecords" />
                </template>
            </cl-card>
        </view>
    </cl-page>
</template>

<script setup lang="ts">
import { ref, computed, onMounted } from 'vue';
import partyService from '@/service/party';

const props = defineProps<{ id: string }>();
const member = ref<PartyMember>({} as PartyMember);
const duesRecords = ref([]);

onMounted(async () => {
    const [detail, dues] = await Promise.all([
        partyService.getMember(props.id),
        partyService.getDuesRecords(props.id),
    ]);
    member.value = detail;
    duesRecords.value = dues.map(d => ({
        title: d.period,
        content: `已缴 ${d.amount_paid} 元`,
        time: d.paid_at,
    }));
});

const memberTypeLabel = computed(() => {
    const map: Record<string, string> = {
        formal: '正式党员',
        probationary: '预备党员',
    };
    return map[member.value.member_type] || '未知';
});
</script>
```

## 5. 登录体系集成

Cool Unix 已内置多种登录方式，只需对接后端：

```typescript
// src/pages/login/index.vue
import { useUserStore } from '@/store/modules/user';

const userStore = useUserStore();

// 微信小程序登录
async function handleWechatLogin() {
    const { code } = await uni.login({ provider: 'weixin' });
    await userStore.loginByWechat(code);
}

// 手机号验证码登录
async function handleSmsLogin() {
    await userStore.loginBySms(mobile.value, smsCode.value);
}

// 单位码进入租户
async function handleOrgCodeLogin() {
    await userStore.loginWithOrgCode(orgCode.value, mobile.value, smsCode.value);
}
```

## 6. SSE 实时推送

```typescript
// src/hooks/useSSE.ts
import { onMounted, onUnmounted } from 'vue';

export function useSSE(onMessage: (event: string, data: any) => void) {
    let eventSource: UniApp.EventSource | null = null;

    onMounted(() => {
        const token = useUserStore().token;
        eventSource = new EventSource(`/api/v1/sse/subscribe?token=${token}`);

        eventSource.addEventListener('notification', (e) => {
            onMessage('notification', JSON.parse(e.data));
        });

        eventSource.addEventListener('approval_status', (e) => {
            onMessage('approval_status', JSON.parse(e.data));
        });
    });

    onUnmounted(() => {
        eventSource?.close();
    });
}
```

## 7. 开发规范

### 7.1 命名约定

| 类型 | 规范 | 示例 |
|------|------|------|
| 页面目录 | 小写中划线 | `party-member-detail/` |
| Vue 组件 | PascalCase | `MemberCard.vue` |
| Composable | camelCase 以 `use` 开头 | `usePartyAuth()` |
| Store | camelCase 以 `use` 开头 + `Store` 后缀 | `useUserStore()` |
| Service 函数 | camelCase 动词开头 | `getMembers()`, `createMeeting()` |
| 类型定义 | PascalCase | `PartyMember`, `PageReq` |

### 7.2 TypeScript 类型定义

```typescript
// src/types/party.d.ts
export interface PartyMember {
    id: string;
    real_name: string;
    gender?: string;
    birth_date?: string;
    mobile?: string;
    join_date?: string;
    member_type: 'formal' | 'probationary' | 'applicant' | 'activist' | 'development_obj';
    position?: string;
    org_name: string;
    avatar_url?: string;
    tags: string[];
    status: string;
}

export interface PageReq {
    page: number;
    page_size: number;
    sort_by?: string;
    sort_order?: 'asc' | 'desc';
}
```
